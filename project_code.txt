"""Structural explain checker.

Reads the five explain_*.json files produced by /admin/explain/... and
verifies each plan contains an IXSCAN in the winning plan, no COLLSCAN
anywhere, and prints examined/returned counts. Exits non-zero on failure.
"""
import json
import pathlib
import sys


def walk(obj):
    """Yield every dict in a nested structure."""
    if isinstance(obj, dict):
        yield obj
        for v in obj.values():
            yield from walk(v)
    elif isinstance(obj, list):
        for item in obj:
            yield from walk(item)


def analyze(explain_doc):
    ix = coll = False
    examined = returned = None
    plan_keys = set()

    # Collect "stage" keys and the specific fields we care about.
    for d in walk(explain_doc):
        stage = d.get("stage")
        if stage == "IXSCAN":
            ix = True
            plan_keys.add("IXSCAN")
        elif stage == "COLLSCAN":
            coll = True
            plan_keys.add("COLLSCAN")
        elif stage:
            plan_keys.add(stage)
        if "totalDocsExamined" in d and examined is None:
            examined = d["totalDocsExamined"]
        if "nReturned" in d and returned is None:
            returned = d["nReturned"]

    return {
        "ixscan": ix,
        "collscan": coll,
        "examined": examined,
        "returned": returned,
        "stages": sorted(plan_keys),
    }


def check(path: str) -> bool:
    try:
        doc = json.loads(pathlib.Path(path).read_text())
    except FileNotFoundError:
        print(f"FAIL {path}: file missing")
        return False
    except json.JSONDecodeError as e:
        print(f"FAIL {path}: invalid JSON ({e})")
        return False

    if "explain" not in doc:
        print(f"FAIL {path}: missing 'explain' key")
        return False

    info = analyze(doc["explain"])
    ok = info["ixscan"] and not info["collscan"]
    status = "OK " if ok else "FAIL"
    examined = info["examined"]
    returned = info["returned"]
    ratio = f" (examined/returned = {examined}/{returned})" if examined is not None and returned else ""
    print(f"{status} {path}: IXSCAN={info['ixscan']} COLLSCAN={info['collscan']}{ratio}")
    print(f"       stages: {info['stages']}")
    return ok


FILES = [
    "explain_attendance_list.json",
    "explain_employee_monthly.json",
    "explain_department_summary.json",
    "explain_late_leaderboard.json",
    "explain_department_trend.json",
]


if __name__ == "__main__":
    results = [check(f) for f in FILES]
    print()
    if all(results):
        print("All explains OK.")
        sys.exit(0)
    print(f"{sum(results)}/{len(results)} passed.")
    sys.exit(1)

# DATA_MODEL.md: what is stored in MongoDB

This file describes the two collections the grader **seeds directly into MongoDB**. Your code must read them exactly as
described here, and what your code writes must look the same. The HTTP contract and the business rules (R1 - R10) live in
`openapi.yaml`; this file refers to rules by number and does not repeat them.

| | |
|---|---|
| Database | the name in `MONGO_DB` |
| Collections | `employees`, `attendance_logs` |
| Indexes | **None** except the default `_id` index. Creating the ones you need, at startup, is your job. |

## 1. Types at a glance

| Kind of value | In MongoDB | In the HTTP API |
|---|---|---|
| An instant (a punch, a creation time, a history time) | BSON date (datetime, UTC) | integer epoch **milliseconds** |
| A calendar date (`date`, `joined_on`) | string `"YYYY-MM-DD"` | the same string |
| A time of day (`shift_start`, `shift_end`) | string `"HH:MM"`, 24-hour, IST | the same string |
| A duration | `work_hours` double, minutes as ints | the same numbers |

Calendar dates and times of day are plain strings **on purpose**: a date is not an instant, and turning it into one forces
a time zone on it. Instants are real datetimes in MongoDB, and your code is responsible for converting them to and from
epoch milliseconds at the API boundary. By default PyMongo returns **naive** datetimes (no time zone); decide how you
will handle that.

## 2. `employees`

One document per employee.

| Field | Type | Notes |
|---|---|---|
| `_id` | ObjectId | **Internal to MongoDB.** Never accepted or returned by the API; do not use it to identify employees. |
| `emp_code` | string | **The employee's identifier everywhere in this assignment.** A customer-generated business code assigned by the customer's HR team, for example `EMP0001`. It must be unique and never changes. It is **not** the `_id`. |
| `name` | string | |
| `email` | string | |
| `department` | string | Case-sensitive. |
| `shift_start` | string `"HH:MM"` | IST. Start of the employee's shift. |
| `shift_end` | string `"HH:MM"` | IST. If `shift_end` <= `shift_start`, the shift is overnight (see R1). |
| `joined_on` | string `"YYYY-MM-DD"` | First day of employment (see R7, R9). |
| `created_at` | BSON date | When the record was created. Set by the API when it creates an employee; never supplied by a client. |

Sample:
```json
{
  "emp_code": "EMP0001",
  "name": "Asha Rao",
  "email": "asha@example.com",
  "department": "Engineering",
  "shift_start": "09:30",
  "shift_end": "18:30",
  "joined_on": "2026-01-05",
  "created_at": {
    "$date": "2026-01-05T04:30:00Z"
  }
}
```

## 3. `attendance_logs`

At most **one document per (`emp_code`, `date`)**. `emp_code` points at `employees.emp_code`. There is no ObjectId reference.

| Field | Type | Notes |
|---|---|---|
| `_id` | ObjectId | Internal. Not part of the API; a record is addressed by (`emp_code`, `date`). |
| `emp_code` | string | The employee. |
| `date` | string `"YYYY-MM-DD"` | The **attendance day**, an IST calendar date, defined by R1. For an overnight shift it is the day the shift started. |
| `status` | string | Exactly one of the five values in the status table below. |
| `punch_in` | BSON date or null | Set at punch-in. Null (or missing) for `ABSENT` / `LEAVE`. |
| `punch_out` | BSON date or null | Null until the employee punches out. |
| `work_hours` | double or null | **Computed by the API**, never supplied by a client (R4). Null until `punch_out` exists. |
| `late_minutes` | int | **Computed by the API** at punch-in (R2). Missing means 0. |
| `overtime_minutes` | int | **Computed by the API** at punch-out (R3). Missing means 0. |
| `half_day` | bool | **Computed by the API** at punch-out (R5). Missing means false. |
| `history` | array | Audit trail of manual corrections. See section 4. Missing means empty. |

### Status values

| `status` | Meaning | Punch times | Counts as present (R6) | Who creates it |
|---|---|---|---|---|
| `PRESENT` | Worked on site | `punch_in` required | yes | Punch-in |
| `WFH` | Worked from home | `punch_in` required | yes | Punch-in |
| `ON_DUTY` | Official duty away from the office | `punch_in` required | yes | Punch-in |
| `ABSENT` | Did not work, not on approved leave | both null | no | Seeded back-office data, or a correction |
| `LEAVE` | On approved leave | both null | no | Seeded back-office data, or a correction |

No endpoint creates `ABSENT` or `LEAVE` records. A weekday with **no document at all** is different from an `ABSENT`
document; the day has simply not been recorded.

### Stored derived values are authoritative

`work_hours`, `late_minutes`, `overtime_minutes` and `half_day` are written by the API, and only by the API, when it
punches in, punches out or corrects a record. Analytics must **read the stored values** and not recompute them from the
punch times. In the grader's seeded data the stored values are what the analytics are defined over.

For a record your own code writes:

| Situation | `work_hours` | `late_minutes` | `overtime_minutes` | `half_day` |
|---|---|---|---|---|
| presence status, no `punch_out` yet | null | from R2 | 0 | false |
| presence status, with `punch_out` | R4 | from R2 | R3 | R5 |
| `ABSENT` or `LEAVE` | null | 0 | 0 | false |

## 4. `history`: the audit trail

`history` records **manual corrections only** (regularizations). Punch-in and punch-out are the original events and never
write to it. Each correction appends exactly one entry; entries are never edited or removed.

| Field | Type | Notes |
|---|---|---|
| `at` | BSON date | When the correction was made. |
| `by` | string | Who made it (free text, no authentication in this assignment). |
| `reason` | string | Why. |
| `changes` | object | One key per field that **actually changed**, each `{ "from": ..., "to": ... }`. Keys can be `status`, `punch_in`, `punch_out`, `work_hours`, `late_minutes`, `overtime_minutes`, `half_day`. |

In MongoDB the `from` / `to` values of `punch_in` and `punch_out` are BSON dates, like the fields themselves. In the API they
are epoch milliseconds. This is easy to forget because the dates sit inside a nested object.

"""Generates ~100k attendance logs and 500 employees for explain testing.
Indexes are created by app/main.py on startup; do NOT create them here."""
import os, random
from datetime import datetime, timedelta, timezone
from dotenv import load_dotenv
from pymongo import MongoClient

load_dotenv()
client = MongoClient(os.getenv("MONGO_URI", "mongodb://localhost:27017"), tz_aware=True)
db = client[os.getenv("MONGO_DB", "attendance_db")]

# Wipe documents only — keeps the indexes that app/main.py created at startup.
db.employees.delete_many({})
db.attendance_logs.delete_many({})

# ---- 500 employees ------------------------------------------------------- #
employees = []
for i in range(1, 501):
    employees.append({
        "emp_code": f"EMP{i:04d}",
        "name": f"Employee {i}",
        "email": f"e{i}@x.com",
        "department": random.choice(["Engineering", "Sales", "Support", "HR", "Finance"]),
        "shift_start": random.choice(["09:30", "10:00", "22:00"]),
        "shift_end":   random.choice(["18:30", "19:00", "06:00"]),
        "joined_on": (datetime(2025, 1, 1) + timedelta(days=random.randint(0, 500))).date().isoformat(),
        "created_at": datetime.now(timezone.utc),
    })
db.employees.insert_many(employees)

# ---- ~100k attendance logs ---------------------------------------------- #
logs = []
base = datetime(2026, 1, 1, tzinfo=timezone.utc)
for e in employees:
    for d in range(200):
        if random.random() < 0.75:
            continue
        day = base + timedelta(days=d)
        status = random.choices(
            ["PRESENT", "WFH", "ON_DUTY", "ABSENT", "LEAVE"],
            weights=[70, 10, 5, 10, 5],
        )[0]
        if status in ("ABSENT", "LEAVE"):
            pi = po = None
            wh = None
            lm = om = 0
            hd = False
        else:
            pi = day.replace(hour=4, minute=random.randint(0, 59), second=0, microsecond=0)
            po = pi + timedelta(hours=random.randint(4, 11), minutes=random.randint(0, 59))
            wh = round((po - pi).total_seconds() / 3600, 2)
            lm = random.choice([0, 0, 0, 0, 5, 15, 30, 45])
            om = random.choice([0, 0, 0, 0, 30, 60, 90])
            hd = wh < 4.5
        logs.append({
            "emp_code": e["emp_code"], "date": day.date().isoformat(),
            "status": status, "punch_in": pi, "punch_out": po,
            "work_hours": wh, "late_minutes": lm, "overtime_minutes": om,
            "half_day": hd, "history": [],
        })

for i in range(0, len(logs), 10_000):
    db.attendance_logs.insert_many(logs[i:i+10_000])

print("attendance_logs:", db.attendance_logs.count_documents({}))

    ## Conventions used by every endpoint
    * **Employees are identified by `emp_code`.** It is a customer-generated business code assigned by the customer's HR
      team (this API does not generate it). It is unique, never changes, and is meant for humans. It is **not** MongoDB's
      `_id`. `_id` is internal: it never appears in a request or a response. Attendance records have no id either; a
      record is addressed by (`emp_code`, `date`).
    * **Instants are epoch milliseconds.** Every point in time in a request or response (`punched_at`, `punch_in`,
      `punch_out`, `created_at`, `history[].at`, and the time values inside `history[].changes`) is an integer count of
      **milliseconds** since 1970-01-01T00:00:00Z. There is no time zone to get wrong. Values that look like seconds
      (below 100000000000), floats, strings and out-of-range numbers are rejected with 422. MongoDB stores these as BSON
      datetimes.
    * **Calendar values are strings.** Dates are `YYYY-MM-DD` (an IST calendar date), months are `YYYY-MM`, shift times
      are `HH:MM` (24-hour, IST).
    * **Derived fields are read-only.** `work_hours`, `late_minutes`, `overtime_minutes` and `half_day` are computed by the
      API from the punch times, the employee's shift and rules R2-R5. They never appear in a request body. If a client
      sends one, ignore it.

    ## Status values
    | `status` | Meaning | Punch times | Counts as present (R6) | Created by |
    |---|---|---|---|---|
    | `PRESENT` | Worked on site | `punch_in` required | yes | punch-in |
    | `WFH` | Worked from home | `punch_in` required | yes | punch-in |
    | `ON_DUTY` | Official duty away from the office | `punch_in` required | yes | punch-in |
    | `ABSENT` | Did not work, no approved leave | both null | no | seeded data, or a correction |
    | `LEAVE` | On approved leave | both null | no | seeded data, or a correction |

    No endpoint creates `ABSENT` or `LEAVE` records. A weekday with **no record at all** is different from an `ABSENT`
    record.

    ## Who computes what
    | Field | Computed when | From |
    |---|---|---|
    | `late_minutes` | punch-in, and after every correction | `punch_in`, shift start (R2) |
    | `work_hours`, `overtime_minutes`, `half_day` | punch-out, and after every correction | `punch_in`, `punch_out`, shift end (R3-R5) |
    | `created_at` | employee creation | server clock |

    For analytics, **read the stored derived values**; do not recompute them from the punch times.

    ## History
    `history` is the audit trail of **manual corrections only**. Punch-in and punch-out never write to it. Each
    successful `PATCH` appends exactly one entry `{at, by, reason, changes}`: when (epoch ms), who (`regularized_by`),
    why (`reason`), and for every field that **actually changed**, `{"from": ..., "to": ...}`. Entries are never edited or
    removed. A record that has never been corrected has `history: []`.

    ## Business rules (R1-R10) - these are NOT visible in the schemas below
    * **R1 Time zone and attendance date.** Shift times and attendance dates are IST (UTC+05:30). A record's `date` is the
      IST calendar date of its `punch_in` instant. Exception - overnight shifts (`shift_end` <= `shift_start`, e.g.
      22:00-06:00): a punch-in whose IST time is *earlier than* `shift_end` belongs to the shift that started the previous
      day. The shift end of an overnight shift falls on the next calendar day. Before any calculation, instants are
      truncated to whole seconds (09:40:00.900 counts as 09:40:00), and the truncated value is what is stored and returned.
    * **R2 Late.** Grace period is 10 minutes. A punch-in is late only if it is *strictly more than* 10:00 after
      `shift_start`. When late, `late_minutes` = whole minutes (floor) elapsed since `shift_start` - not since the end of
      the grace period. Otherwise 0. Example (shift 09:30): 09:40:00 -> 0; 09:40:01 -> 10; 10:15:59 -> 45.
    * **R3 Overtime.** `overtime_minutes` = whole minutes (floor) between `shift_end` and `punch_out`, counted only when
      that is >= 30; otherwise 0.
    * **R4 Work hours.** (`punch_out` - `punch_in`) in seconds / 3600, rounded to 2 decimals **half-up** (not banker's
      rounding, not float rounding). `null` until punch-out.
    * **R5 Half day.** After punch-out, `half_day` = true when the **rounded** `work_hours` < 4.50. A half day counts
      as 0.5 of a present day.
    * **R6 Presence.** `PRESENT`, `WFH` and `ON_DUTY` count as present. `ABSENT` and `LEAVE` do not.
    * **R7 Working days.** Mon-Fri, no holiday calendar. For an employee who joined mid-month, working days are counted
      from `joined_on`. `present_days` only counts records that fall on Mon-Fri (weekend records still contribute to
      late and overtime totals).
    * **R8 Rounding of reported numbers.** 2 decimals half-up (rates: 4 decimals half-up).
    * **R9 Headcount.** Employees whose `joined_on` is on or before the end of the period (the last day of the month, or
      the given day for the trend), whether or not they have any logs.
    * **R10 Pagination.** `page` starts at 1. `page_size` max is 100 (default 20). `total` always reflects the applied filters.

def to_ist(dt: datetime) -> datetime:
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    return dt.astimezone(IST)


def truncate_seconds(dt: datetime) -> datetime:
    if dt is None:
        return None
    return dt.replace(microsecond=0)


def epoch_ms(dt) -> Optional[int]:
    if dt is None:
        return None
    if isinstance(dt, (int, float)):
        return int(dt)
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    return int(dt.timestamp() * 1000)


def from_epoch_ms(ms: Optional[int]) -> Optional[datetime]:
    if ms is None:
        return None
    return datetime.fromtimestamp(ms / 1000, tz=timezone.utc)


from typing import Annotated
from pydantic import AfterValidator

def _validate_date_str(v: str) -> str:
    if not isinstance(v, str) or len(v) != 10 or v[4] != "-" or v[7] != "-":
        raise ValueError("date must be YYYY-MM-DD")
    try:
        date_cls.fromisoformat(v)
    except ValueError:
        raise ValueError("date must be a valid calendar date")
    # Reject things like 2026-13-01 that fromisoformat might accept on some versions
    y, m, d = v.split("-")
    if not (1 <= int(m) <= 12) or not (1 <= int(d) <= 31):
        raise ValueError("date must be a valid calendar date")
    return v

DateStr = Annotated[str, AfterValidator(_validate_date_str)]

# Rounding (half-up)
# --------------------------------------------------------------------------- #
def round_half_up(value: Optional[float], places: int) -> Optional[float]:
    if value is None:
        return None
    q = Decimal(10) ** -places
    return float(Decimal(str(value)).quantize(q, rounding=ROUND_HALF_UP))

# Business rules R1 - R5
# --------------------------------------------------------------------------- #
def compute_attendance_date(punch_in: datetime, shift_start: str, shift_end: str) -> str:
    """R1: IST calendar date; overnight shifts that start 'late' belong to previous day."""
    pi = truncate_seconds(to_ist(punch_in))
    sh, sm = map(int, shift_start.split(":"))
    eh, em = map(int, shift_end.split(":"))
    shift_start_min = sh * 60 + sm
    shift_end_min = eh * 60 + em
    d = pi.date()
    if shift_end_min <= shift_start_min:
        punch_min = pi.hour * 60 + pi.minute
        if punch_min < shift_end_min:
            d = d - timedelta(days=1)
    return d.isoformat()


def compute_late_minutes(punch_in: datetime, shift_start: str, attendance_date: str) -> int:
    """R2: grace 10 min; late iff strictly more than 10:00 past shift_start."""
    pi = truncate_seconds(to_ist(punch_in))
    d = date_cls.fromisoformat(attendance_date)
    sh, sm = map(int, shift_start.split(":"))
    start = datetime(d.year, d.month, d.day, sh, sm, 0, 0, tzinfo=IST)
    diff = (pi - start).total_seconds()
    if diff > 600:
        return int(diff // 60)
    return 0


def compute_work_hours(punch_in: datetime, punch_out: datetime) -> float:
    """R4: (out - in) seconds / 3600, half-up to 2 decimals."""
    pi = truncate_seconds(to_ist(punch_in))
    po = truncate_seconds(to_ist(punch_out))
    hours = (po - pi).total_seconds() / 3600.0
    return round_half_up(hours, 2)


def compute_overtime_minutes(punch_out: datetime, shift_end: str, attendance_date: str, shift_start: str) -> int:
    """R3: whole minutes from shift_end to punch_out, only when >= 30."""
    d = date_cls.fromisoformat(attendance_date)
    sh, sm = map(int, shift_start.split(":"))
    eh, em = map(int, shift_end.split(":"))
    end_date = d
    if (eh * 60 + em) <= (sh * 60 + sm):
        end_date = d + timedelta(days=1)
    end = datetime(end_date.year, end_date.month, end_date.day, eh, em, 0, 0, tzinfo=IST)
    po = truncate_seconds(to_ist(punch_out))
    diff_min = int((po - end).total_seconds() // 60)
    return diff_min if diff_min >= 30 else 0


def compute_half_day(work_hours: Optional[float]) -> bool:
    """R5: true when rounded work_hours < 4.50."""
    if work_hours is None:
        return False
    return work_hours < 4.50

def serialize_history(hist: List[dict]) -> List[dict]:
    out = []
    for h in hist or []:
        changes = {}
        for k, v in (h.get("changes") or {}).items():
            if k in ("punch_in", "punch_out"):
                changes[k] = {"from": epoch_ms(v.get("from")), "to": epoch_ms(v.get("to"))}
            else:
                changes[k] = {"from": v.get("from"), "to": v.get("to")}
        out.append({
            "at": epoch_ms(h["at"]),
            "by": h["by"],
            "reason": h["reason"],
            "changes": changes,
        })
    return out


def serialize_attendance(doc: dict) -> dict:
    return {
        "emp_code": doc["emp_code"],
        "date": doc["date"],
        "status": doc["status"],
        "punch_in": epoch_ms(doc.get("punch_in")),
        "punch_out": epoch_ms(doc.get("punch_out")),
        "work_hours": doc.get("work_hours"),
        "late_minutes": doc.get("late_minutes", 0),
        "overtime_minutes": doc.get("overtime_minutes", 0),
        "half_day": bool(doc.get("half_day", False)),
        "history": serialize_history(doc.get("history", [])),
    }

def _validate_epoch(v):
    """Rejects strings, floats, bools, out-of-range ints. Called in mode='before'."""
    if v is None:
        return v
    if isinstance(v, bool) or not isinstance(v, int):
        raise ValueError("epoch milliseconds must be an integer")
    if v < EPOCH_MIN or v > EPOCH_MAX:
        raise ValueError("epoch milliseconds out of range")

@app.post("/attendance/punch-in", status_code=201)
def punch_in(body: PunchInIn):
    emp = db.employees.find_one({"emp_code": body.emp_code})
    if not emp:
        raise HTTPException(status_code=404, detail="Employee not found")

    ts = from_epoch_ms(body.punched_at) if body.punched_at is not None else datetime.now(timezone.utc)
    ts = truncate_seconds(ts)

    att_date = compute_attendance_date(ts, emp["shift_start"], emp["shift_end"])
    late = compute_late_minutes(ts, emp["shift_start"], att_date)

    doc = {
        "emp_code": body.emp_code,
        "date": att_date,
        "status": body.status.value,
        "punch_in": ts,
        "punch_out": None,
        "work_hours": None,
        "late_minutes": late,
        "overtime_minutes": 0,
        "half_day": False,
        "history": [],
    }
    try:
        db.attendance_logs.insert_one(doc)
    except DuplicateKeyError:
        raise HTTPException(status_code=409, detail="already punched in for this date")
    return serialize_attendance(doc)


@app.post("/attendance/punch-out")
def punch_out(body: PunchOutIn):
    emp = db.employees.find_one({"emp_code": body.emp_code})
    if not emp:
        raise HTTPException(status_code=404, detail="Employee not found")

    ts = from_epoch_ms(body.punched_at) if body.punched_at is not None else datetime.now(timezone.utc)
    ts = truncate_seconds(ts)

    rec = db.attendance_logs.find_one(
        {"emp_code": body.emp_code, "punch_in": {"$lte": ts}, "punch_out": None},
        sort=[("punch_in", DESCENDING)],
    )
    if not rec:
        raise HTTPException(status_code=404, detail="No open punch-in found")

    pi = rec["punch_in"]
    if pi.tzinfo is None:
        pi = pi.replace(tzinfo=timezone.utc)

    diff = (ts - pi).total_seconds()
    if diff <= 0 or diff > 24 * 3600:
        raise HTTPException(status_code=422, detail="Invalid punch_out (must be after punch_in and within 24h)")

    work_hours = compute_work_hours(pi, ts)
    overtime = compute_overtime_minutes(ts, emp["shift_end"], rec["date"], emp["shift_start"])
    half_day = compute_half_day(work_hours)

    # Atomic update — only succeeds if still open (race-safe)
    result = db.attendance_logs.update_one(
        {"_id": rec["_id"], "punch_out": None},
        {"$set": {
            "punch_out": ts,
            "work_hours": work_hours,
            "overtime_minutes": overtime,
            "half_day": half_day,
        }},
    )
    if result.modified_count == 0:
        raise HTTPException(status_code=409, detail="Record already punched out")

    updated = db.attendance_logs.find_one({"_id": rec["_id"]})
    return serialize_attendance(updated)

@app.patch("/attendance/{emp_code}/{date}")
def regularize(
    emp_code: str = Path(...),
    date: DateStr = Path(...),
    body: RegularizeIn = ...,
):
    emp = db.employees.find_one({"emp_code": emp_code})
    if not emp:
        raise HTTPException(status_code=404, detail="Employee not found")

    rec = db.attendance_logs.find_one({"emp_code": emp_code, "date": date})
    if not rec:
        raise HTTPException(status_code=404, detail="Record not found")

    # ---- Determine effective new state -----------------------------------
    eff_status = body.status.value if body.status else rec["status"]

    if eff_status in ("ABSENT", "LEAVE"):
        if body.punch_in is not None or body.punch_out is not None:
            raise HTTPException(status_code=422, detail="Cannot supply punch times with ABSENT/LEAVE")
        eff_pi_ms = None
        eff_po_ms = None
    else:
        # Always canonicalise to a whole second: convert -> truncate -> re-encode.
        raw_pi = body.punch_in if body.punch_in is not None else epoch_ms(rec.get("punch_in"))
        if raw_pi is None:
            raise HTTPException(status_code=422, detail="Presence status requires a punch_in")
        pi_dt = truncate_seconds(from_epoch_ms(raw_pi))
        eff_pi_ms = epoch_ms(pi_dt)  # canonical ms, used for storage and history

        if compute_attendance_date(pi_dt, emp["shift_start"], emp["shift_end"]) != date:
            raise HTTPException(status_code=422, detail="punch_in must remain on the record's date")

        raw_po = body.punch_out if body.punch_out is not None else epoch_ms(rec.get("punch_out"))
        if raw_po is not None:
            po_dt = truncate_seconds(from_epoch_ms(raw_po))
            eff_po_ms = epoch_ms(po_dt)  # canonical ms, used for storage and history
            d = (po_dt - pi_dt).total_seconds()
            if d <= 0 or d > 24 * 3600:
                raise HTTPException(status_code=422, detail="punch_out must be after punch_in and within 24h")
        else:
            eff_po_ms = None

    # ---- Recompute derived fields ---------------------------------------
    if eff_status in ("ABSENT", "LEAVE"):
        new_late, new_wh, new_ot, new_hd = 0, None, 0, False
    else:
        pi_dt = truncate_seconds(from_epoch_ms(eff_pi_ms))
        new_late = compute_late_minutes(pi_dt, emp["shift_start"], date)
        if eff_po_ms is not None:
            po_dt = truncate_seconds(from_epoch_ms(eff_po_ms))
            new_wh = compute_work_hours(pi_dt, po_dt)
            new_ot = compute_overtime_minutes(po_dt, emp["shift_end"], date, emp["shift_start"])
            new_hd = compute_half_day(new_wh)
        else:
            new_wh, new_ot, new_hd = None, 0, False

    # ---- Diff against stored values -------------------------------------
    old_status = rec["status"]
    old_pi_ms = epoch_ms(rec.get("punch_in"))
    old_po_ms = epoch_ms(rec.get("punch_out"))
    old_wh = rec.get("work_hours")
    old_late = rec.get("late_minutes", 0)
    old_ot = rec.get("overtime_minutes", 0)
    old_hd = bool(rec.get("half_day", False))

    changes: dict = {}
    def _c(field, old, new):
        if old != new:
            changes[field] = {"from": old, "to": new}

    _c("status", old_status, eff_status)
    _c("punch_in", old_pi_ms, eff_pi_ms)
    _c("punch_out", old_po_ms, eff_po_ms)
    _c("work_hours", old_wh, new_wh)
    _c("late_minutes", old_late, new_late)
    _c("overtime_minutes", old_ot, new_ot)
    _c("half_day", old_hd, new_hd)

    if not changes:
        raise HTTPException(status_code=422, detail="No changes")

    # ---- Build history entry (BSON dates for punch times) ---------------
    hist_changes: dict = {}
    for k, v in changes.items():
        if k in ("punch_in", "punch_out"):
            hist_changes[k] = {
                "from": from_epoch_ms(v["from"]) if v["from"] is not None else None,
                "to": from_epoch_ms(v["to"]) if v["to"] is not None else None,
            }
        else:
            hist_changes[k] = {"from": v["from"], "to": v["to"]}

    hist_entry = {
        "at": datetime.now(timezone.utc),
        "by": body.regularized_by,
        "reason": body.reason,
        "changes": hist_changes,
    }

    # ---- Optimistic update ----------------------------------------------
    result = db.attendance_logs.update_one(
        {
            "_id": rec["_id"],
            "status": old_status,
            "punch_in": rec.get("punch_in"),
            "punch_out": rec.get("punch_out"),
        },
        {
            "$set": {
                "status": eff_status,
                "punch_in": from_epoch_ms(eff_pi_ms),
                "punch_out": from_epoch_ms(eff_po_ms),
                "work_hours": new_wh,
                "late_minutes": new_late,
                "overtime_minutes": new_ot,
                "half_day": new_hd,
            },
            "$push": {"history": hist_entry},
        },
    )
    if result.modified_count == 0:
        raise HTTPException(status_code=409, detail="Concurrent modification")

    updated = db.attendance_logs.find_one({"_id": rec["_id"]})
    return serialize_attendance(updated)

def _month_bounds(month: str):
    y, m = map(int, month.split("-"))
    first = date_cls(y, m, 1)
    if m == 12:
        last = date_cls(y, 12, 31)
    else:
        last = date_cls(y, m + 1, 1) - timedelta(days=1)
    return first, last


def _count_working_days(start: date_cls, end: date_cls) -> int:
    n = 0
    d = start
    while d <= end:
        if d.weekday() < 5:
            n += 1
        d += timedelta(days=1)
    return n

@app.get("/analytics/employees/{emp_code}/monthly")
def employee_monthly(
    emp_code: str = Path(...),
    month: str = Query(..., pattern=r"^\d{4}-(0[1-9]|1[0-2])$"),
):
    emp = db.employees.find_one({"emp_code": emp_code})
    if not emp:
        raise HTTPException(status_code=404, detail="Employee not found")

    first, last = _month_bounds(month)
    joined = date_cls.fromisoformat(emp["joined_on"])
    wd_start = max(first, joined)
    working_days = _count_working_days(wd_start, last) if wd_start <= last else 0

    date_from, date_to = first.isoformat(), last.isoformat()

    pipeline = [
        {"$match": {"emp_code": emp_code, "date": {"$gte": date_from, "$lte": date_to}}},
        {"$addFields": {
            "dow": {"$dayOfWeek": {"$dateFromString": {"dateString": "$date"}}},
        }},
        {"$group": {
            "_id": None,
            "leave_days": {"$sum": {"$cond": [{"$eq": ["$status", "LEAVE"]}, 1, 0]}},
            "late_count": {"$sum": {"$cond": [{"$gt": [{"$ifNull": ["$late_minutes", 0]}, 0]}, 1, 0]}},
            "total_late_minutes": {"$sum": {"$ifNull": ["$late_minutes", 0]}},
            "total_overtime_minutes": {"$sum": {"$ifNull": ["$overtime_minutes", 0]}},
            "present_days": {"$sum": {
                "$cond": [
                    {"$and": [
                        {"$in": ["$status", ["PRESENT", "WFH", "ON_DUTY"]]},
                        {"$gte": ["$dow", 2]},
                        {"$lte": ["$dow", 6]},
                    ]},
                    {"$cond": [{"$ifNull": ["$half_day", False]}, 0.5, 1]},
                    0,
                ]
            }},
        }},
    ]
    res = list(db.attendance_logs.aggregate(pipeline))
    r = res[0] if res else {
        "leave_days": 0, "late_count": 0, "total_late_minutes": 0,
        "total_overtime_minutes": 0, "present_days": 0,
    }

    present_days = float(r["present_days"])
    attendance_pct = None if working_days == 0 else round_half_up(present_days / working_days * 100, 2)

    return {
        "emp_code": emp_code,
        "month": month,
        "working_days": working_days,
        "present_days": present_days,
        "leave_days": int(r["leave_days"]),
        "late_count": int(r["late_count"]),
        "total_late_minutes": int(r["total_late_minutes"]),
        "total_overtime_minutes": int(r["total_overtime_minutes"]),
        "attendance_pct": attendance_pct,
    }

def _dept_summary_pipeline(department: Optional[str], date_from: str, date_to: str):
    emp_match: dict = {"joined_on": {"$lte": date_to}}
    if department:
        emp_match["department"] = department

    wh_match = {
        "$and": [
            {"$in": ["$status", ["PRESENT", "WFH", "ON_DUTY"]]},
            {"$ne": [{"$ifNull": ["$work_hours", None]}, None]},
        ]
    }
    present_match = {
        "$and": [
            {"$in": ["$status", ["PRESENT", "WFH", "ON_DUTY"]]},
            {"$gte": [{"$dayOfWeek": {"$dateFromString": {"dateString": "$date"}}}, 2]},
            {"$lte": [{"$dayOfWeek": {"$dateFromString": {"dateString": "$date"}}}, 6]},
        ]
    }

    return [
        # Start with employees so zero-log employees are counted.
        {"$match": emp_match},
        # For each employee, compute a single stats document (never a log array).
        {"$lookup": {
            "from": "attendance_logs",
            "let": {"ec": "$emp_code"},
            "pipeline": [
                {"$match": {
                    "$expr": {"$eq": ["$emp_code", "$$ec"]},
                    "date": {"$gte": date_from, "$lte": date_to},
                }},
                {"$group": {
                    "_id": None,
                    "present_days": {"$sum": {"$cond": [
                        present_match,
                        {"$cond": [{"$ifNull": ["$half_day", False]}, 0.5, 1]},
                        0,
                    ]}},
                    "late_count": {"$sum": {"$cond": [
                        {"$gt": [{"$ifNull": ["$late_minutes", 0]}, 0]}, 1, 0,
                    ]}},
                    "total_late_minutes": {"$sum": {"$ifNull": ["$late_minutes", 0]}},
                    "leave_count": {"$sum": {"$cond": [
                        {"$eq": ["$status", "LEAVE"]}, 1, 0,
                    ]}},
                    "on_duty_count": {"$sum": {"$cond": [
                        {"$eq": ["$status", "ON_DUTY"]}, 1, 0,
                    ]}},
                    "wh_sum": {"$sum": {"$cond": [wh_match, "$work_hours", 0]}},
                    "wh_count": {"$sum": {"$cond": [wh_match, 1, 0]}},
                }},
            ],
            "as": "stats",
        }},
        # Flatten the 0/1-element array from the lookup.
        {"$addFields": {
            "stats": {"$ifNull": [
                {"$arrayElemAt": ["$stats", 0]},
                {
                    "present_days": 0, "late_count": 0, "total_late_minutes": 0,
                    "leave_count": 0, "on_duty_count": 0, "wh_sum": 0, "wh_count": 0,
                },
            ]},
        }},
        # Sum per-employee stats into per-department stats.
        {"$group": {
            "_id": "$department",
            "headcount": {"$sum": 1},
            "present_days": {"$sum": "$stats.present_days"},
            "late_count": {"$sum": "$stats.late_count"},
            "total_late_minutes": {"$sum": "$stats.total_late_minutes"},
            "leave_count": {"$sum": "$stats.leave_count"},
            "on_duty_count": {"$sum": "$stats.on_duty_count"},
            "wh_sum": {"$sum": "$stats.wh_sum"},
            "wh_count": {"$sum": "$stats.wh_count"},
        }},
        {"$addFields": {
            "avg_work_hours": {"$cond": [
                {"$gt": ["$wh_count", 0]},
                {"$divide": ["$wh_sum", "$wh_count"]},
                None,
            ]},
        }},
        {"$project": {
            "_id": 0, "department": "$_id", "headcount": 1, "present_days": 1,
            "avg_work_hours": 1, "late_count": 1, "total_late_minutes": 1,
            "leave_count": 1, "on_duty_count": 1,
        }},
        {"$sort": {"department": 1}},
    ]

# Analytics — leaderboard
# --------------------------------------------------------------------------- #
def _leaderboard_pipeline(month: str, limit: int, department: Optional[str]):
    first, last = _month_bounds(month)
    date_from, date_to = first.isoformat(), last.isoformat()

    pipeline = [
        {"$match": {
            "date": {"$gte": date_from, "$lte": date_to},
            "late_minutes": {"$gt": 0},
        }},
        {"$group": {
            "_id": "$emp_code",
            "total_late_minutes": {"$sum": "$late_minutes"},
            "late_count": {"$sum": 1},
        }},
        {"$lookup": {
            "from": "employees",
            "localField": "_id",
            "foreignField": "emp_code",
            "as": "emp",
        }},
        {"$unwind": "$emp"},
    ]
    if department:
        pipeline.append({"$match": {"emp.department": department}})

    pipeline.extend([
        {"$setWindowFields": {
            "sortBy": {"total_late_minutes": -1},
            "output": {"rank": {"$rank": {}}},
        }},
        {"$match": {"rank": {"$lte": limit}}},
        {"$sort": {"total_late_minutes": -1, "_id": 1}},
        {"$project": {
            "_id": 0,
            "rank": 1,
            "emp_code": "$_id",
            "name": "$emp.name",
            "department": "$emp.department",
            "total_late_minutes": 1,
            "late_count": 1,
        }},
    ])
    return pipeline

# --------------------------------------------------------------------------- #
# Analytics — department trend
# --------------------------------------------------------------------------- #
def _trend_pipeline(department: str, from_: str, to: str, emp_codes: list, joined_dates: list,
                    from_dt: datetime, to_dt: datetime):
    densify_upper = to_dt + timedelta(days=1)
    return [
        {"$match": {
            "emp_code": {"$in": emp_codes},
            "date": {"$gte": from_, "$lte": to},
        }},
        {"$group": {
            "_id": "$date",
            "present_count": {"$sum": {"$cond": [
                {"$in": ["$status", ["PRESENT", "WFH", "ON_DUTY"]]},
                {"$cond": [{"$ifNull": ["$half_day", False]}, 0.5, 1]},
                0,
            ]}},
            "late_count": {"$sum": {"$cond": [
                {"$gt": [{"$ifNull": ["$late_minutes", 0]}, 0]}, 1, 0,
            ]}},
        }},
        {"$addFields": {"date_obj": {"$dateFromString": {"dateString": "$_id"}}}},
        {"$densify": {
            "field": "date_obj",
            "range": {"step": 1, "unit": "day", "bounds": [from_dt, densify_upper]},
        }},
        {"$addFields": {
            "date": {"$dateToString": {"date": "$date_obj", "format": "%Y-%m-%d"}},
            "present_count": {"$ifNull": ["$present_count", 0]},
            "late_count": {"$ifNull": ["$late_count", 0]},
        }},
        {"$sort": {"date_obj": 1}},
        {"$addFields": {
            "is_working_day": {"$let": {
                "vars": {"dow": {"$dayOfWeek": "$date_obj"}},
                "in": {"$and": [{"$gte": ["$$dow", 2]}, {"$lte": ["$$dow", 6]}]},
            }},
            "headcount": {"$size": {"$filter": {
                "input": {"$literal": joined_dates},
                "as": "j",
                "cond": {"$lte": ["$$j", "$date"]},
            }}},
        }},
        {"$addFields": {
            "attendance_rate": {"$cond": [
                {"$and": ["$is_working_day", {"$gt": ["$headcount", 0]}]},
                {"$divide": ["$present_count", "$headcount"]},
                None,
            ]},
        }},
        {"$setWindowFields": {
            "sortBy": {"date_obj": 1},
            "output": {"moving_avg_7d": {
                "$avg": "$attendance_rate",
                "window": {"documents": [-6, 0]},
            }},
        }},
        {"$project": {
            "_id": 0, "date": 1, "is_working_day": 1, "headcount": 1,
            "present_count": 1, "late_count": 1, "attendance_rate": 1, "moving_avg_7d": 1,
        }},
    ]


@app.get("/analytics/departments/{department}/trend")
def department_trend(
    department: str = Path(...),
    from_: str = Query(..., alias="from"),
    to: str = Query(...),
):
    try:
        d_from = date_cls.fromisoformat(from_)
        d_to = date_cls.fromisoformat(to)
    except ValueError:
        raise HTTPException(status_code=422, detail="Invalid date format")

    if d_to < d_from:
        raise HTTPException(status_code=422, detail="to must be >= from")
    if (d_to - d_from).days > 91:
        raise HTTPException(status_code=422, detail="Range must not exceed 92 days")

    emps = list(db.employees.find(
        {"department": department},
        {"_id": 0, "emp_code": 1, "joined_on": 1},
    ))
    if not emps:
        raise HTTPException(status_code=404, detail="Department not found")

    emp_codes = [e["emp_code"] for e in emps]
    joined_dates = [e["joined_on"] for e in emps]
    from_dt = datetime(d_from.year, d_from.month, d_from.day, tzinfo=timezone.utc)
    to_dt = datetime(d_to.year, d_to.month, d_to.day, tzinfo=timezone.utc)

    pipeline = _trend_pipeline(department, from_, to, emp_codes, joined_dates, from_dt, to_dt)
    items = list(db.attendance_logs.aggregate(pipeline))

    for it in items:
        it["headcount"] = int(it["headcount"])
        it["present_count"] = float(it["present_count"])
        it["late_count"] = int(it["late_count"])
        if it["attendance_rate"] is not None:
            it["attendance_rate"] = round_half_up(it["attendance_rate"], 4)
        if it["moving_avg_7d"] is not None:
            it["moving_avg_7d"] = round_half_up(it["moving_avg_7d"], 4)

    return {"department": department, "items": items}

            if date_to:
                rng["$lte"] = date_to
            q["date"] = rng
        if status:
            q["status"] = status.value
        raw = _explain({
            "explain": {
                "find": "attendance_logs",
                "filter": q,
                "sort": {"date": -1, "emp_code": 1},
                "skip": (page - 1) * page_size,
                "limit": page_size,

            "verbosity": "executionStats",
        })
        return {"endpoint": endpoint, "collection": "attendance_logs",
                "explain": json.loads(json_util.dumps(raw))}

    if endpoint == "employee_monthly":
        if not emp_code or not month:
            raise HTTPException(status_code=422, detail="emp_code and month required")

        date_from_, date_to_ = first.isoformat(), last.isoformat()
        pipeline = [

                            {"$cond": [{"$ifNull": ["$half_day", False]}, 0.5, 1]},
                            0,

        ]
        raw = _explain({
            "explain": {"aggregate": "attendance_logs", "pipeline": pipeline, "cursor": {}},
            "verbosity": "executionStats",
        })
        return {"endpoint": endpoint, "collection": "attendance_logs",
                "explain": json.loads(json_util.dumps(raw))}

    if endpoint == "department_summary":
        if not month:
            raise HTTPException(status_code=422, detail="month required")
        first, last = _month_bounds(month)
        pipeline = _dept_summary_pipeline(department, first.isoformat(), last.isoformat())

        pipeline = _leaderboard_pipeline(month, limit, department)
        raw = _explain({
            "explain": {"aggregate": "attendance_logs", "pipeline": pipeline, "cursor": {}},
            "verbosity": "executionStats",
        })
        return {"endpoint": endpoint, "collection": "attendance_logs",
                "explain": json.loads(json_util.dumps(raw))}

    if endpoint == "department_trend":
        if not department or not from_ or not to:
            raise HTTPException(status_code=422, detail="department, from and to required")
        emps = list(db.employees.find({"department": department}, {"_id": 0, "emp_code": 1, "joined_on": 1}))
        emp_codes = [e["emp_code"] for e in emps]
        joined_dates = [e["joined_on"] for e in emps]
        d_from = date_cls.fromisoformat(from_)
        d_to = date_cls.fromisoformat(to)
        from_dt = datetime(d_from.year, d_from.month, d_from.day, tzinfo=timezone.utc)
        to_dt = datetime(d_to.year, d_to.month, d_to.day, tzinfo=timezone.utc)
        pipeline = _trend_pipeline(department, from_, to, emp_codes, joined_dates, from_dt, to_dt)
        raw = _explain({
            "explain": {"aggregate": "attendance_logs", "pipeline": pipeline, "cursor": {}},
            "verbosity": "executionStats",
        })
        return {"endpoint": endpoint, "collection": "attendance_logs",
                "explain": json.loads(json_util.dumps(raw))}

    raise HTTPException(status_code=422, detail="Unknown endpoint")
